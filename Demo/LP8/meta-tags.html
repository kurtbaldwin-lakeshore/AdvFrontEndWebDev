<head>
  <!-- Controls the browser tab AND the search result's clickable headline -->
  <title>Meta Tags — Web Trends, SEO & Accessibility</title>

  <!-- Controls the gray snippet text under the search result -->
  <meta name="description"
        content="See exactly which meta tag controls the browser tab, the search snippet, and the social media preview.">
</head>
