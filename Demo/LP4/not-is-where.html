style.css 
/* :not() — matches everything that does NOT match */
.niw-list li:not(:last-child) {
  border-bottom: 2px solid #dbe7ea;
}

/* :is() — groups selectors, same as writing each rule separately */
:is(.niw-heading, .niw-alt) {
  color: #a65314;
  font-family: 'Consolas', monospace;
}

/* :where() — same grouping, but ALWAYS zero specificity */
:where(.niw-list) li { padding: 0.4rem 0; }
