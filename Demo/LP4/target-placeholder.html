style.css 
/* :target — styles the element whose id matches the URL fragment */
#tp-section:target {
  background: #fef3c7;
  scroll-margin-top: 2rem;
}

/* ::placeholder — styles placeholder text in inputs/textareas */
.tp-input::placeholder {
  color: #a65314;
  font-style: italic;
}
