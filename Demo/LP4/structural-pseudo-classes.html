style.css 
.spc-table tr:first-child { font-weight: 700; }

/* One specific position */
.spc-table tr:nth-child(2) { color: #a65314; }

/* Zebra striping — the classic nth-child use case */
.spc-table tr:nth-child(odd)  { background: #f2f6f7; }
.spc-table tr:nth-child(even) { background: #fff; }
