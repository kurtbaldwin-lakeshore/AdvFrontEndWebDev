index.html 
<div class="shape ft-base">Original</div>
<div class="shape ft-blur">blur(3px)</div>
<div class="shape ft-bright">brightness(1.4)</div>
<div class="shape ft-gray">Hover me</div>
style.css 
.ft-blur   { filter: blur(3px); }
.ft-bright { filter: brightness(1.4); }

/* Hover effect: full color fades in from grayscale */
.ft-gray { filter: grayscale(100%); }
.ft-gray:hover,
.ft-gray:focus-visible { filter: grayscale(0%); }
