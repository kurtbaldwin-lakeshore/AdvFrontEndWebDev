style.css 
.bs-flat { border-radius: 10px; }

.bs-single {
  border-radius: 10px;
  box-shadow: 0 10px 15px rgba(0,0,0,.3);
}

/* Tight dark shadow close in, plus a soft diffuse shadow */
/* further out — this combination reads as REAL depth */
.bs-layered {
  border-radius: 10px;
  box-shadow:
    0 2px 4px rgba(0,0,0,.25),
    0 12px 24px rgba(0,0,0,.18);
}
